from flask import Flask, request, jsonify
from threading import Lock

app = Flask(__name__)

cart = {
    "Pear": 0,
    "Strawberry": 0,
    "Pineapple": 0,
    "Orange": 0
}

cart_lock = Lock()


def normalize_fruit(fruit):
    if not fruit:
        return None

    fruit = str(fruit).strip().lower()

    fruit_map = {
        "pear": "Pear",
        "strawberry": "Strawberry",
        "strawberries": "Strawberry",
        "pineapple": "Pineapple",
        "pine apple": "Pineapple",
        "pine-apple": "Pineapple",
        "orange": "Orange"
    }

    return fruit_map.get(fruit)


@app.route("/")
def home():
    return jsonify({
        "status": "Fruit Billing API is running",
        "cart": cart
    })


@app.route("/status", methods=["GET"])
def status():
    with cart_lock:
        return jsonify({
            "cart": cart
        })


@app.route("/add_item", methods=["POST"])
def add_item():
    data = request.get_json(silent=True)

    if not data:
        return jsonify({
            "success": False,
            "error": "JSON data required"
        }), 400

    fruit = normalize_fruit(data.get("fruit"))

    if fruit is None:
        return jsonify({
            "success": False,
            "error": "Unknown fruit"
        }), 400

    with cart_lock:
        cart[fruit] += 1

        return jsonify({
            "success": True,
            "fruit": fruit,
            "count": cart[fruit],
            "cart": cart
        })


@app.route("/remove_item", methods=["POST"])
def remove_item():
    data = request.get_json(silent=True)

    if not data:
        return jsonify({
            "success": False,
            "error": "JSON data required"
        }), 400

    fruit = normalize_fruit(data.get("fruit"))

    if fruit is None:
        return jsonify({
            "success": False,
            "error": "Unknown fruit"
        }), 400

    with cart_lock:
        if cart[fruit] > 0:
            cart[fruit] -= 1

        return jsonify({
            "success": True,
            "fruit": fruit,
            "count": cart[fruit],
            "cart": cart
        })


@app.route("/clear", methods=["POST"])
def clear_cart():
    with cart_lock:
        for fruit in cart:
            cart[fruit] = 0

    return jsonify({
        "success": True,
        "message": "Cart cleared",
        "cart": cart
    })


if __name__ == "__main__":
    import os

    port = int(os.environ.get("PORT", 5000))

    app.run(
        host="0.0.0.0",
        port=port
    )
